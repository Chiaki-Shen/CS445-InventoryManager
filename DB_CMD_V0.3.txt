create table zipLocations(
  zipCode INT NOT NULL,
  city VARCHAR(30),
  stateAbbr VARCHAR(2) NOT NULL,
  CONSTRAINT zipLocations_pk PRIMARY KEY (zipCode),
  CONSTRAINT state_zipLocations_fk FOREIGN KEY (stateAbbr) REFERENCES states(abbreviation)
);

create table vendors(
  vendorName VARCHAR(50) NOT NULL,
  phoneNumber VARCHAR(20),
  email VARCHAR(50),
  streetAddress VARCHAR (50),
  zipCode INT,
  websiteURL VARCHAR(50),
  CONSTRAINT vendors_pk PRIMARY KEY (vendorName),
  CONSTRAINT vendors_ck UNIQUE (phoneNumber),
  CONSTRAINT vendors_ck2 UNIQUE (email),
  CONSTRAINT zipLocations_vendors_fk FOREIGN KEY (zipCode) REFERENCES zipLocations(zipCode)
);

create table products(
  SKU VARCHAR (20) NOT NULL,
  category VARCHAR (50) default 'N/A',
  prodDesc VARCHAR NOT NULL,
  productURL VARCHAR,
  quantity VARCHAR (10) NOT NULL,
  availability VARCHAR(10) NOT NULL,
  --unitsInStock double NOT NULL,
  CONSTRAINT products_pk PRIMARY KEY (SKU),
  CONSTRAINT categories_products_fk FOREIGN KEY (category) REFERENCES categories(categoryName)
);

create table inventory(
    inventory_id char(36) default (lower(hex(randomblob(4))) || '-' || lower(hex(randomblob(2))) || '-4' || substr(lower(hex(randomblob(2))),2) || '-' || substr('89ab',abs(random()) % 4 + 1, 1) || substr(lower(hex(randomblob(2))),2) || '-' || lower(hex(randomblob(6)))),
    SKU VARCHAR (25) NOT NULL,
    unitsInStock double,
    CONSTRAINT inventory_pk PRIMARY KEY (inventory_id),
    CONSTRAINT product_inventory_fk FOREIGN KEY (SKU) REFERENCES products (SKU)
);

create table vendorPrices(
    vendorPriceID char(36) default (lower(hex(randomblob(4))) || '-' || lower(hex(randomblob(2))) || '-4' || substr(lower(hex(randomblob(2))),2) || '-' || substr('89ab',abs(random()) % 4 + 1, 1) || substr(lower(hex(randomblob(2))),2) || '-' || lower(hex(randomblob(6)))),
    vendorName VARCHAR(50) default 'N/A',
    SKU VARCHAR(15) NOT NULL,
    unitListPrice DOUBLE NOT NULL,
    timeChecked DateTime DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT vendorPrices_pk PRIMARY KEY (vendorPriceID),
    CONSTRAINT products_vendorPrices_fk FOREIGN KEY (SKU) REFERENCES products(SKU),
    CONSTRAINT vendors_vendorPrices_fk FOREIGN KEY (vendorName) REFERENCES vendors(vendorName)
);

create table status(
  status VARCHAR(10) NOT NULL,
  CONSTRAINT status_pk PRIMARY KEY (status)
);

CREATE TABLE orders(
    orderId char(36) default (lower(hex(randomblob(4))) || '-' || lower(hex(randomblob(2))) || '-4' || substr(lower(hex(randomblob(2))),2) || '-' || substr('89ab',abs(random()) % 4 + 1, 1) || substr(lower(hex(randomblob(2))),2) || '-' || lower(hex(randomblob(6)))),
    orderDate DATETIME,
    vendorName VARCHAR (50) NOT NULL,
    orderStatus VARCHAR(10) NOT NULL,
    CONSTRAINT orderID_pk PRIMARY KEY (orderID),
    CONSTRAINT orders_vendorPrice_fk FOREIGN KEY (vendorName) REFERENCES vendorprices(vendorName),
    CONSTRAINT status_vendorPrice_fk FOREIGN KEY (orderStatus) REFERENCES status(status)
);

--use unitSalePrice for subtotal(ext price tx) and order total
CREATE TABLE orderLines(
    orderId char(36) default (lower(hex(randomblob(4))) || '-' || lower(hex(randomblob(2))) || '-4' || substr(lower(hex(randomblob(2))),2) || '-' || substr('89ab',abs(random()) % 4 + 1, 1) || substr(lower(hex(randomblob(2))),2) || '-' || lower(hex(randomblob(6)))),
    SKU VARCHAR (15) NOT NULL,
    productDesc VARCHAR (255),
    quantity double NOT NULL,
    unitSalePrice double NOT NULL,
    vendorPriceID char(36) default (lower(hex(randomblob(4))) || '-' || lower(hex(randomblob(2))) || '-4' || substr(lower(hex(randomblob(2))),2) || '-' || substr('89ab',abs(random()) % 4 + 1, 1) || substr(lower(hex(randomblob(2))),2) || '-' || lower(hex(randomblob(6)))),
    CONSTRAINT orderLines_pk PRIMARY KEY (orderID, SKU),
    CONSTRAINT orders_orderLines_fk FOREIGN KEY (orderID) REFERENCES orders(orderID),
    CONSTRAINT vendorPrices_orderLines_fk FOREIGN KEY (vendorPriceID) REFERENCES vendorprices(vendorPriceID),
    CONSTRAINT products_orderLines_fk FOREIGN KEY (SKU) REFERENCES products (SKU)
);
