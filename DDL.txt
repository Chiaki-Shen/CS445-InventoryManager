create table zipLocations(
  zipCode INT NOT NULL,
  city VARCHAR(30),
  stateAbbr VARCHAR(2) NOT NULL,
  CONSTRAINT zipLocations_pk PRIMARY KEY (zipCode),
  CONSTRAINT state_zipLocations_fk FOREIGN KEY (stateAbbr) REFERENCES states(abbreviation)
);

create table vendors(
  vendorName VARCHAR(50) NOT NULL,
  phoneNumber VARCHAR(20),
  email VARCHAR(50),
  streetAddress VARCHAR (50),
  websiteURL VARCHAR(50),
  zipCode INT,
  CONSTRAINT vendors_pk PRIMARY KEY (vendorName),
  CONSTRAINT vendors_ck UNIQUE (phoneNumber),
  CONSTRAINT vendors_ck2 UNIQUE (email),
  CONSTRAINT zipLocations_vendors_fk FOREIGN KEY (zipCode) REFERENCES zipLocations(zipCode)
);

create table products(
  SKU VARCHAR (25) NOT NULL,
  category VARCHAR (50) DEFAULT 'N/A',
  prodDesc VARCHAR NOT NULL,
  productURL VARCHAR,
  quantity VARCHAR (10) NOT NULL,
  availability VARCHAR(10) NOT NULL,
  reorderLevel double DEFAULT 0,
  unitsInStock double DEFAULT 0,
  CONSTRAINT products_pk PRIMARY KEY (SKU),
  CONSTRAINT categories_products_fk FOREIGN KEY (category) REFERENCES categories(categoryName)
);


create table vendorPrices(
    vendorPriceID INTEGER NOT NULL,
    vendorName VARCHAR(50),
    SKU VARCHAR(25) NOT NULL,
    unitListPrice DOUBLE NOT NULL,
    timeChecked DateTime DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT vendorPrices_pk PRIMARY KEY (vendorPriceID),
    CONSTRAINT products_vendorPrices_fk FOREIGN KEY (SKU) REFERENCES products(SKU),
    CONSTRAINT vendors_vendorPrices_fk FOREIGN KEY (vendorName) REFERENCES vendors(vendorName)
);

create table status(
  status VARCHAR(10) NOT NULL,
  CONSTRAINT status_pk PRIMARY KEY (status)
);

CREATE TABLE orders(
    orderId VARCHAR (50) NOT NULL,
    orderDate DATETIME,
    orderStatus VARCHAR(10) NOT NULL,
    CONSTRAINT orderID_pk PRIMARY KEY (orderID),
    CONSTRAINT status_vendorPrice_fk FOREIGN KEY (orderStatus) REFERENCES status(status)
);

SELECT o.orderId, o.orderDate, oL.SKU, o.orderStatus
    FROM orders o
    INNER JOIN orderLines oL on o.orderId = oL.orderId
    INNER JOIN status s on o.orderStatus = s.status
    INNER JOIN products p on p.SKU = oL.SKU
    INNER JOIN vendorPrices vP on oL.SKU = vP.SKU
GROUP BY o.orderID, oL.SKU
ORDER BY o.orderDate DESC;
