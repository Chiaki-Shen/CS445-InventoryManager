  zipCode INT NOT NULL,

  CONSTRAINT zipLocations_pk PRIMARY KEY (zipCode),
  CONSTRAINT state_zipLocations_fk FOREIGN KEY (stateAbbr) REFERENCES states(abbreviation)

create table vendors(
  vendorName VARCHAR(50) NOT NULL,
  phoneNumber VARCHAR(20),
  email VARCHAR(50),
  streetAddress VARCHAR (50),
  websiteURL VARCHAR(50),
  zipCode INT,
  CONSTRAINT vendors_pk PRIMARY KEY (vendorName),
  CONSTRAINT vendors_ck UNIQUE (phoneNumber),
  CONSTRAINT vendors_ck2 UNIQUE (email),
  CONSTRAINT zipLocations_vendors_fk FOREIGN KEY (zipCode) REFERENCES zipLocations(zipCode)
);

create table vendorPrices(
    vendorPriceID INTEGER NOT NULL,

--- ORDERLINE SUBTOTAL
SELECT oL.SKU, p.prodDesc, unitListPrice, itemQuantity,
       (unitListPrice * itemQuantity) AS subtotal
    FROM orders o
        INNER JOIN orderLines oL on o.orderId = oL.orderId
        INNER JOIN products p on p.SKU = oL.SKU
        INNER JOIN vendorPrices vP on p.SKU = vP.SKU
WHERE o.orderId = '010a5635-c0f3-496a-b20f-397d6cdbfa83'
GROUP BY o.orderID, oL.SKU;


-- Show Order Total Sum
SELECT SUM(unitListPrice * itemQuantity)
    FROM orders o
        INNER JOIN orderLines oL on o.orderId = oL.orderId
        INNER JOIN products p on p.SKU = oL.SKU
        INNER JOIN vendorPrices vP on p.SKU = vP.SKU
WHERE o.orderId = '010a5635-c0f3-496a-b20f-397d6cdbfa83'
GROUP BY o.orderID;

--- ORDER TOTALS
SELECT o.orderID, o.orderDate, o.orderStatus,
        (SELECT COUNT(oL.sku)
            FROM orderLines oL
        WHERE o.orderId =oL.orderId )AS number_of_items,
       ROUND(SUM(unitListPrice * oL.itemQuantity)/2, 2) AS total
    FROM orders o
        INNER JOIN orderLines oL on o.orderId = oL.orderId
        INNER JOIN products p on p.SKU = oL.SKU
        INNER JOIN vendorPrices vP on p.SKU = vP.SKU
GROUP BY o.orderID, o.orderDate
